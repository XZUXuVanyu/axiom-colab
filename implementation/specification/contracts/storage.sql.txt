-- Specification for schema1. These statements become the two migration files named in the manifest.
-- IDs and hashes TEXT; revisions constrained INTEGER >=1 (V1 max SQLite signed int64).
-- Payload JSON validated by C++ record serializer; no trusted decision read from payload alone.
-- WORKSPACE DATABASE
CREATE TABLE workspace(id TEXT PRIMARY KEY, name TEXT NOT NULL, created_at TEXT NOT NULL);
CREATE TABLE goal(id TEXT PRIMARY KEY, workspace_id TEXT NOT NULL REFERENCES workspace(id), revision INTEGER NOT NULL CHECK(revision>=1), state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE plan(id TEXT PRIMARY KEY, goal_id TEXT NOT NULL REFERENCES goal(id), revision INTEGER NOT NULL CHECK(revision>=1), hash TEXT NOT NULL UNIQUE, state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE project(id TEXT PRIMARY KEY, workspace_id TEXT NOT NULL REFERENCES workspace(id), revision INTEGER NOT NULL CHECK(revision>=1), payload TEXT NOT NULL);
CREATE TABLE revision_proposal(id TEXT PRIMARY KEY, project_id TEXT NOT NULL REFERENCES project(id), baseline_hash TEXT NOT NULL, content_hash TEXT NOT NULL, state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE candidate(id TEXT PRIMARY KEY, project_id TEXT NOT NULL REFERENCES project(id), content_hash TEXT NOT NULL UNIQUE, state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE validation(id TEXT PRIMARY KEY, candidate_id TEXT NOT NULL REFERENCES candidate(id), binary_manifest_hash TEXT NOT NULL, suite_hash TEXT NOT NULL, conclusion TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE approval(id TEXT PRIMARY KEY, subject_kind TEXT NOT NULL, subject_hash TEXT NOT NULL, decision TEXT NOT NULL, consumed_by TEXT UNIQUE, reviewer_session TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE binding(id TEXT PRIMARY KEY, workspace_id TEXT NOT NULL REFERENCES workspace(id), package_hash TEXT NOT NULL, revision INTEGER NOT NULL CHECK(revision>=1), grant_json TEXT NOT NULL, revoked INTEGER NOT NULL CHECK(revoked IN(0,1)), UNIQUE(workspace_id,package_hash));
CREATE TABLE object_revision(object_id TEXT NOT NULL, revision INTEGER NOT NULL CHECK(revision>=1), kind TEXT NOT NULL, content_hash TEXT NOT NULL, type_id TEXT NOT NULL, payload TEXT NOT NULL, PRIMARY KEY(object_id,revision));
CREATE TABLE job(id TEXT PRIMARY KEY, kind TEXT NOT NULL, state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE graph_revision(graph_id TEXT NOT NULL, revision INTEGER NOT NULL CHECK(revision>=1), semantic_hash TEXT NOT NULL, graph_json TEXT NOT NULL, view_json TEXT NOT NULL, PRIMARY KEY(graph_id,revision));
CREATE TABLE run(id TEXT PRIMARY KEY, goal_id TEXT NOT NULL REFERENCES goal(id), plan_hash TEXT NOT NULL REFERENCES plan(hash), graph_hash TEXT NOT NULL, state TEXT NOT NULL, snapshot_json TEXT NOT NULL);
CREATE TABLE run_node(run_id TEXT NOT NULL REFERENCES run(id), node_path TEXT NOT NULL, job_id TEXT REFERENCES job(id), state TEXT NOT NULL, payload TEXT NOT NULL, PRIMARY KEY(run_id,node_path));
CREATE TABLE audit(seq INTEGER PRIMARY KEY AUTOINCREMENT, utc TEXT NOT NULL, actor_kind TEXT NOT NULL, operation TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE idempotency(authority_scope TEXT NOT NULL, operation TEXT NOT NULL, request_key TEXT NOT NULL, request_hash TEXT NOT NULL, response_json TEXT NOT NULL, PRIMARY KEY(authority_scope,operation,request_key));
CREATE TABLE outbox(id TEXT PRIMARY KEY, operation TEXT NOT NULL, subject_hash TEXT NOT NULL, state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE private_suite(id TEXT PRIMARY KEY, content_hash TEXT NOT NULL, private_object_ref TEXT NOT NULL, configured_by TEXT NOT NULL);
PRAGMA user_version=1;

-- LIBRARY DATABASE (a separate file; not executed together with workspace statements)
CREATE TABLE package(hash TEXT PRIMARY KEY, tool_id TEXT NOT NULL, version TEXT NOT NULL, state TEXT NOT NULL, manifest_json TEXT NOT NULL, UNIQUE(tool_id,version));
CREATE TABLE package_file(package_hash TEXT NOT NULL REFERENCES package(hash), path TEXT NOT NULL, content_hash TEXT NOT NULL, bytes INTEGER NOT NULL CHECK(bytes>=0), role TEXT NOT NULL, PRIMARY KEY(package_hash,path));
CREATE TABLE dependency(package_hash TEXT NOT NULL REFERENCES package(hash), dependency_hash TEXT NOT NULL REFERENCES package(hash), service_id TEXT NOT NULL, PRIMARY KEY(package_hash,dependency_hash,service_id));
CREATE TABLE publication(id TEXT PRIMARY KEY, package_hash TEXT NOT NULL REFERENCES package(hash), approval_digest TEXT NOT NULL, state TEXT NOT NULL, payload TEXT NOT NULL);
CREATE TABLE revocation(package_hash TEXT PRIMARY KEY REFERENCES package(hash), utc TEXT NOT NULL, reason TEXT NOT NULL);
PRAGMA user_version=1;
-- Adapters enforce state enums and object existence before writes; SQL parameters are always bound.
-- No generic query command is exposed to model/UI. Library publication order respects dependency FKs.
